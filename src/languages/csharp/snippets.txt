=== WordCount.cs
static Dictionary<string, int> WordCount(string text)
{
    var counts = new Dictionary<string, int>();
    foreach (var word in text.Split(' ', StringSplitOptions.RemoveEmptyEntries))
    {
        counts[word] = counts.GetValueOrDefault(word) + 1;
    }
    return counts;
}

{

    foreach (var row in rows)
    {
        sb.AppendLine(string.Join(",", row));

=== BankAccount.cs
public class BankAccount
{
    private readonly List<decimal> _transactions = [];

    public BankAccount(string owner)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(owner);
        Owner = owner;
    }

    public string Owner { get; }
    public decimal Balance => _transactions.Sum();

    public void Deposit(decimal amount)
    {
        if (amount <= 0)
        {
            throw new ArgumentOutOfRangeException(nameof(amount), "Must be positive.");
        }
        _transactions.Add(amount);
    }

    public void Withdraw(decimal amount)
    {
        if (amount > Balance)
        {
            throw new InvalidOperationException("Insufficient funds.");
        }
        _transactions.Add(-amount);
    }
}
=== Repository.cs
public interface IRepository<T> where T : class
{
    T? Find(int id);
    IReadOnlyList<T> All();
    void Save(int id, T item);
    bool Delete(int id);
}

public class InMemoryRepository<T> : IRepository<T> where T : class
{
    private readonly Dictionary<int, T> _items = new();

    public T? Find(int id) => _items.GetValueOrDefault(id);

    public IReadOnlyList<T> All() => _items.Values.ToList();

    public void Save(int id, T item) => _items[id] = item;

    public bool Delete(int id) => _items.Remove(id);
}
=== OrderService.cs
public record OrderLine(string Product, int Quantity, decimal Price);

public record Order(int Id, List<OrderLine> Lines)
{
    public decimal Total => Lines.Sum(l => l.Quantity * l.Price);
}

public interface IOrderStore
{
    Task<IReadOnlyList<Order>> AllAsync(CancellationToken token);
}

public class OrderService(IOrderStore store)
{
    public async Task<decimal> RevenueAsync(CancellationToken token = default)
    {
        var orders = await store.AllAsync(token);
        return orders.Sum(o => o.Total);
    }

    public async Task<Order?> LargestAsync(CancellationToken token = default)
    {
        var orders = await store.AllAsync(token);
        return orders.MaxBy(o => o.Total);
    }
}
=== Retry.cs
public static class Retry
{
    public static async Task<T> RunAsync<T>(
        Func<CancellationToken, Task<T>> action,
        int attempts = 3,
        CancellationToken token = default)
    {
        var delay = TimeSpan.FromMilliseconds(200);
        for (var attempt = 1; ; attempt++)
        {
            try
            {
                return await action(token);
            }
            catch (HttpRequestException) when (attempt < attempts)
            {
                await Task.Delay(delay, token);
                delay *= 2;
            }
        }
    }
}
=== StepRange.cs
using System.Collections;

public class StepRange(int start, int end, int step = 1) : IEnumerable<int>
{
    public IEnumerator<int> GetEnumerator()
    {
        for (var i = start; i < end; i += step)
        {
            yield return i;
        }
    }

    IEnumerator IEnumerable.GetEnumerator() => GetEnumerator();
}
