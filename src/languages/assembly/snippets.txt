=== echo.asm
section .bss
    buffer resb 64

section .text
    global _start

_start:
    ; read up to 64 bytes from standard input
    xor eax, eax
    xor edi, edi
    mov rsi, buffer
    mov rdx, 64
    syscall
    test rax, rax
    jle .exit

    ; write them back to standard output
    mov rdx, rax
    mov rax, 1
    mov rdi, 1
    mov rsi, buffer
    syscall
    jmp _start

.exit:
    mov rax, 60
    xor edi, edi
    syscall
=== to_upper.asm
; void to_upper(char *text): uppercases ASCII letters in place
global to_upper
to_upper:
    mov al, [rdi]
    test al, al
    jz .done
    cmp al, 'a'
    jb .next
    cmp al, 'z'
    ja .next
    sub al, 32
    mov [rdi], al
.next:
    inc rdi
    jmp to_upper
.done:
    ret
=== reverse.asm
; void reverse(char *text, long length): reverses text in place
global reverse
reverse:
    lea rsi, [rdi + rsi - 1]
.swap:
    cmp rdi, rsi
    jae .done
    mov al, [rdi]
    mov dl, [rsi]
    mov [rdi], dl
    mov [rsi], al
    inc rdi
    dec rsi
    jmp .swap
.done:
    ret

    digits resb 21

    global _start

    mov rax, 2026
    call print_number
    mov rax, 60
    xor edi, edi
    syscall

    lea rsi, [digits + 20]
    mov byte [rsi], 10
    mov rcx, 10
.next:
    xor edx, edx
    div rcx
    add dl, '0'
    dec rsi
    mov [rsi], dl
    test rax, rax
    jnz .next

    mov rax, 1
    mov rdi, 1
    lea rdx, [digits + 21]
    sub rdx, rsi
    syscall
    ret
=== bubble_sort.asm
; void bubble_sort(long *values, long count)
global bubble_sort
bubble_sort:
    dec rsi
    jle .done
.pass:
    xor ecx, ecx
    xor r8d, r8d
.compare:
    mov rax, [rdi + rcx * 8]
    mov rdx, [rdi + rcx * 8 + 8]
    cmp rax, rdx
    jle .ordered
    mov [rdi + rcx * 8], rdx
    mov [rdi + rcx * 8 + 8], rax
    mov r8d, 1
.ordered:
    inc rcx
    cmp rcx, rsi
    jl .compare
    test r8d, r8d
    jnz .pass
.done:
    ret
