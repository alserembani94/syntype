use std::collections::HashMap;

fn word_count(text: &str) -> HashMap<&str, usize> {
    let mut counts = HashMap::new();
    for word in text.split_whitespace() {
        *counts.entry(word).or_insert(0) += 1;
    }
    counts
}

fn average(values: &[f64]) -> Option<f64> {
    if values.is_empty() {
        return None;
    }
    let total: f64 = values.iter().sum();
    Some(total / values.len() as f64)
}

=== shapes.rs
trait Shape {
    fn area(&self) -> f64;
    fn name(&self) -> String;
}

struct Circle {
    radius: f64,
}

struct Rect {
    width: f64,
    height: f64,
}

impl Shape for Circle {
    fn area(&self) -> f64 {
        std::f64::consts::PI * self.radius * self.radius
    }

    fn name(&self) -> String {
        String::from("circle")
    }
}

impl Shape for Rect {
    fn area(&self) -> f64 {
        self.width * self.height
    }

    fn name(&self) -> String {
        String::from("rect")
    }
}

pub struct Stack<T> {
    items: Vec<T>,
}

impl<T> Stack<T> {
    pub fn new() -> Self {
        Stack { items: Vec::new() }
    }

    pub fn push(&mut self, item: T) {
        self.items.push(item);
    }

    pub fn pop(&mut self) -> Option<T> {
        self.items.pop()
    }

    pub fn peek(&self) -> Option<&T> {
        self.items.last()
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }
}

use std::collections::HashMap;
use std::fs;

#[derive(Debug)]
enum ConfigError {
    Io(std::io::Error),
    Missing(String),
}

impl From<std::io::Error> for ConfigError {
    fn from(err: std::io::Error) -> Self {
        ConfigError::Io(err)
    }
}

fn load(path: &str) -> Result<HashMap<String, String>, ConfigError> {
    let text = fs::read_to_string(path)?;
    let pairs: HashMap<String, String> = text
        .lines()
        .filter_map(|line| line.split_once("="))
        .map(|(k, v)| (k.trim().to_string(), v.trim().to_string()))
        .collect();
    if !pairs.contains_key("name") {
        return Err(ConfigError::Missing("name".into()));
    }
    Ok(pairs)
}

use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let counter = Arc::new(Mutex::new(0));
    let mut handles = vec![];

    for _ in 0..8 {
        let counter = Arc::clone(&counter);
        handles.push(thread::spawn(move || {
            *counter.lock().unwrap() += 1;
        }));
    }

    for handle in handles {
        handle.join().unwrap();
    }

    println!("Result: {}", *counter.lock().unwrap());
}
=== inventory.rs
#[derive(Debug, Clone)]
struct Item {
    name: String,
    price: u32,
    quantity: u32,
}

struct Inventory {
    items: Vec<Item>,
}

impl Inventory {
    fn total_value(&self) -> u32 {
        self.items.iter().map(|item| item.price * item.quantity).sum()
    }

    fn low_stock(&self, threshold: u32) -> Vec<&Item> {
        self.items
            .iter()
            .filter(|item| item.quantity < threshold)
            .collect()
    }
}
